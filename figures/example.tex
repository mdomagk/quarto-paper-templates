% example figure of template.qmd: a damped oscillation
%   pdflatex example.tex
\documentclass[border=2pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[width=8.5cm, height=5cm, xlabel={Time in s}, ylabel={Voltage in p.u.},
               grid=major, xmin=0, xmax=10, samples=200, legend pos=north east]
    \addplot[thick, blue, domain=0:10] {exp(-0.3*x)*cos(deg(3*x))};
    \addplot[dashed, gray, domain=0:10] {exp(-0.3*x)};
    \legend{Signal, Envelope}
  \end{axis}
\end{tikzpicture}
\end{document}
